## 📄 INSTRUCTIONS.md

# EchoMind - Running & Testing Guide

## 🚀 Running the Application (Development)
1. **Activate your virtual environment**:
   ```bash
   # Windows
   venv\Scripts\activate
   # Mac/Linux
   source venv/bin/activate
   ```

2. **Start the Flask backend**:
   ```bash
   cd backend
   python app.py
   ```

3. **Start the React frontend**:
   ```bash
   cd frontend
   npm run dev
   ```

4. **Open your browser at the React URL** (usually):
   ```
   http://localhost:5173
   ```

✅ React handles the UI, Flask handles API routes (`/upload`, `/stream/<job_id>`, `/download/<job_id>`). Both must run at the same time in development.

---

## ✅ Running Automated Tests (Backend)
1. **Enable Test Mode**:
   ```powershell
   # Windows PowerShell
   set ECHO_TEST_MODE=1
   # Mac/Linux
   export ECHO_TEST_MODE=1
   ```

2. **Start the backend in test mode**:
   ```bash
   python app.py
   ```

3. **In another terminal, run the tester**:
   ```bash
   python tests/test_backend_v2.py
   ```

4. **View reports** in `tests/reports/`.

---

## 🛠 Deployment (Production)
1. **Install backend dependencies**:
   ```bash
   pip install -r requirements.txt
   ```

2. **Build React frontend**:
   ```bash
   cd frontend
   npm run build
   ```

3. **Serve built files with Flask or Nginx**:
   - Copy `frontend/dist` into a static folder or configure Flask to serve it.
   - Start backend with Gunicorn:
     ```bash
     gunicorn -w 4 app:app
     ```

4. (Optional) Use Nginx as a reverse proxy for SSL and static assets.

---

    reports/            # Test reports (contents ignored in snapshots)

---

## 🔑 Notes:
- `@cross_origin()` is enabled on `/upload`, `/stream`, `/download` routes for React ↔ Flask communication.
- React URL (`localhost:5173`) and Flask URL (`127.0.0.1:5000`) must both be running during development.
- For production, React is built and served from Flask or a reverse proxy like Nginx.
- `logs/`, `temp/`, and `tests/reports/` are kept for context but their file contents are ignored in snapshots.

        path = os.path.join(TEMP_DIR, item)
        try:
            if os.path.isdir(path):
                shutil.rmtree(path, ignore_errors=True)
            else:
                os.remove(path)
        except Exception as e:
            logger.error(f"Startup cleanup error on {path}: {e}")

def cleanup_old_temp():
    try:
        total_size = 0
        folder_stats = []
        for root, dirs, files in os.walk(TEMP_DIR):
            for f in files:
                total_size += os.path.getsize(os.path.join(root, f))
            for d in dirs:
                folder_path = os.path.join(root, d)
                folder_size = sum(os.path.getsize(os.path.join(dp, file))
                                  for dp, dn, filenames in os.walk(folder_path) for file in filenames)
                last_modified = os.path.getmtime(folder_path)
                folder_stats.append((folder_path, folder_size, last_modified))

        if total_size / (1024 * 1024) > MAX_DISK_USAGE_MB:
            logger.warning("Disk usage exceeded, purging old temp folders...")
            for folder_path, _, _ in sorted(folder_stats, key=lambda x: (x[2], x[1])):
                folder_uuid = os.path.basename(folder_path)
                if folder_uuid in active_processing or folder_uuid in active_downloading:
                    continue
                shutil.rmtree(folder_path, ignore_errors=True)
                logger.info(f"Purged job folder: {folder_path}")
    except Exception as e:
        logger.error(f"Cleanup error: {e}")

def process_job(job_id, upload_path, filename, selected_stems):
    start_time = time.time()
    try:
        log(job_id, "Starting Demucs split...")

        demucs_dir = os.path.join(TEMP_DIR, job_id, "stems")
        preview_dir = os.path.join(TEMP_DIR, job_id, "preview")
        os.makedirs(preview_dir, exist_ok=True)
        base_name = os.path.splitext(filename)[0]
        zip_output_path = os.path.join(TEMP_DIR, job_id, f"{base_name}_stems.zip")

        use_two_stems = sorted(selected_stems) == ["other", "vocals"]
        track_dir = split_audio(
            upload_path,
            output_dir=demucs_dir,
            stem_type="2stems" if use_two_stems else "all",
            job_id=job_id,
            logger=logger
        )

        all_possible_stems = {"vocals", "drums", "bass", "other", "no_vocals"}
        for stem in all_possible_stems:
            if stem == "no_vocals":
                if "other" in selected_stems:
                    continue
            elif stem not in selected_stems:
                file_path = os.path.join(track_dir, f"{stem}.wav")
                if os.path.exists(file_path):
                    os.remove(file_path)

        for stem in selected_stems:
            src = os.path.join(track_dir, f"{stem}.wav")
            if stem == "other" and not os.path.exists(src):
                fallback = os.path.join(track_dir, "no_vocals.wav")
                if os.path.exists(fallback):
                    src = fallback
            if os.path.exists(src):
                dst_preview = os.path.join(preview_dir, f"{stem}.mp3")
                audio = AudioSegment.from_wav(src)
                export_opts = {"format": "mp3", "bitrate": "192k"}
                if len(audio) > 30_000:
                    audio[:30_000].export(dst_preview, **export_opts)
                else:
                    audio.export(dst_preview, **export_opts)

        shutil.make_archive(zip_output_path.replace(".zip", ""), 'zip', track_dir)

        try:
            os.remove(upload_path)
        except Exception:
            pass
        shutil.rmtree(demucs_dir, ignore_errors=True)

        with jobs_lock:
            JOBS[job_id]["status"] = "done"
            JOBS[job_id]["zip"] = zip_output_path

        duration = time.time() - start_time
        log(job_id, f"Processing time: {duration:.2f} seconds")
        log(job_id, "Job complete. ZIP ready.")
        JOBS[job_id]["events"].put("Job complete")

        def auto_cleanup():
            time.sleep(JOB_EXPIRATION)
            with jobs_lock:
                if job_id in JOBS and JOBS[job_id]["status"] == "done":
                    folder = os.path.join(TEMP_DIR, job_id)
                    shutil.rmtree(folder, ignore_errors=True)
                    JOBS.pop(job_id, None)
                    logger.info(f"[CLEANUP] Auto-deleted job {job_id}")

        threading.Thread(target=auto_cleanup, daemon=True).start()

    except Exception as e:
        with jobs_lock:
            if job_id in JOBS:
                JOBS[job_id]["status"] = "error"
                JOBS[job_id]["zip"] = None
        log(job_id, f"Error: {e}")
    finally:
        active_processing.discard(job_id)



def is_rate_limited(ip):
    now = time.time()
    window = 60
    max_uploads = 2  # Your .env value

    with jobs_lock:
        history = UPLOAD_HISTORY[ip]

        # Remove expired timestamps
        while history and now - history[0] > window:
            history.popleft()

        # ✅ BLOCK if over frequency limit
        if len(history) >= max_uploads:
            return f"Upload rate limit exceeded ({max_uploads} per {window}s)."

        # ✅ Count this attempt no matter what
        history.append(now)

        # ✅ Then check active jobs
        active_jobs = sum(
            1 for job in JOBS.values()
            if job.get("ip") == ip and job.get("status") == "processing"
        )
        if active_jobs >= MAX_JOBS_PER_IP:
            return f"Too many jobs processing (limit: {MAX_JOBS_PER_IP})."

    return None

    file_size_mb = round(file.tell() / (1024 * 1024), 2)
    file.seek(0)

    user_id = None
    is_guest = True

        try:
            conn = sqlite3.connect(db_path)
            c = conn.cursor()

                user_id = result[0]

        except Exception as e:
            print("Failed to look up user:", e)
        finally:
            if conn:
                conn.close()

    # 👇 Usage check
    if is_guest:
        weekly_usage = get_guest_weekly_usage(ip)
        cap = 150
    else:
        weekly_usage = get_account_weekly_usage(user_id)
        cap = 500

    if weekly_usage + file_size_mb > cap:
        return jsonify({
            "error": f"Weekly limit exceeded ({cap}MB max). You’ve used {round(weekly_usage, 2)}MB."
        }), 403

    selected_stems = request.form.getlist("stems") or ["vocals", "other"]
    job_id = str(uuid.uuid4())
    active_processing.add(job_id)

    temp_upload_dir = os.path.join(TEMP_DIR, job_id)
    os.makedirs(temp_upload_dir, exist_ok=True)
    upload_path = os.path.join(temp_upload_dir, filename)
    file.save(upload_path)

    try:
        duration = float(mediainfo(upload_path).get("duration", "0"))
        if duration <= 0:
            raise ValueError("No valid audio stream")
        if duration > 600:
            raise ValueError("File too long")
    except Exception:
        shutil.rmtree(temp_upload_dir, ignore_errors=True)
        active_processing.discard(job_id)
        return jsonify({"error": "Invalid or corrupted audio file."}), 400

    if Path(filename).suffix.lower() == ".mp4":
        wav_path = os.path.join(temp_upload_dir, f"{Path(filename).stem}.wav")
        cmd = ["ffmpeg", "-i", upload_path, "-vn", "-acodec", "pcm_s16le", wav_path, "-y"]
        log(job_id, f"Extracting audio from MP4: {' '.join(cmd)}")
        result = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        if result.returncode != 0 or not os.path.exists(wav_path) or os.path.getsize(wav_path) < 4096:
            shutil.rmtree(temp_upload_dir, ignore_errors=True)
            active_processing.discard(job_id)
            return jsonify({"error": "Invalid or corrupted audio file."}), 400
        upload_path = wav_path
        log(job_id, "Extracted audio from MP4.")

    # 👇 Log usage to DB
    try:
        conn = sqlite3.connect(db_path)
        c = conn.cursor()
        upload_date = time.strftime("%Y-%m-%d")
        c.execute(
            "INSERT INTO usage (user_id, ip, date, mb_used) VALUES (?, ?, ?, ?)",
            (user_id, ip, upload_date, file_size_mb)
        )
        conn.commit()
    except Exception as e:
        print("Failed to log usage:", e)
    finally:
        if conn:
            conn.close()

    with jobs_lock:
        JOBS[job_id] = {"status": "processing", "zip": None, "events": queue.Queue(), "ip": ip}

    threading.Thread(target=process_job, args=(job_id, upload_path, filename, selected_stems)).start()
    return jsonify({"job_id": job_id})

        yield "data: Processing started...\n\n"
        while True:

"""
audio_splitter.py
-----------------
Handles audio separation using Demucs.
Runs Demucs completely silent except for start and completion logs.

import subprocess
import os

def split_audio(input_path, output_dir="separated", stem_type="all", job_id=None, logger=None, model_name="htdemucs"):
    """
    Split an audio file into stems using Demucs silently.

    Args:
        input_path (str): Path to the audio file to split.
        output_dir (str): Directory where separated stems are saved.
        stem_type (str): Type of separation ('all' or '2stems').
        job_id (str): Optional job ID for log prefixing.
        logger (logging.Logger): Optional Flask logger.

    Returns:
        str: Path to the directory containing separated stems.
    """
def split_audio(input_path, output_dir="separated", stem_type="all", job_id=None, logger=None):
    """
    Split an audio file into stems using Demucs silently.

    Args:
        input_path (str): Path to the audio file to split.
        output_dir (str): Directory where separated stems are saved.
        stem_type (str): Type of separation ('all' or '2stems').
        job_id (str): Optional job ID for log prefixing.
        logger (logging.Logger): Optional Flask logger.

    Returns:
        str: Path to the directory containing separated stems.
    """
    os.makedirs(output_dir, exist_ok=True)
    model_args = []

    if stem_type == "2stems":
        model_args = ["--two-stems", "vocals"]

    command = ["demucs", input_path, "-o", output_dir] + model_args

    if logger:
        logger.info(f"[{job_id}] Demucs process starting...")
    else:
        print(f"[{job_id}] Demucs process starting...")

    subprocess.run(command, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=True)

    if logger:
        logger.info(f"[{job_id}] Demucs process complete.")
    else:
        print(f"[{job_id}] Demucs process complete.")

    base_name = os.path.splitext(os.path.basename(input_path))[0]
    model_output_dir = os.listdir(output_dir)[0]
    output_path = os.path.join(output_dir, model_output_dir, base_name)

    if logger:
        logger.info(f"[{job_id}] Demucs wrote to: {os.path.join(output_dir, model_output_dir)}")

    return output_path

## 📄 frontend/src/api.js
import axios from "axios";

const API_BASE = "http://127.0.0.1:5000"; // Flask dev server URL

export async function uploadFile(file, selectedStems, onProgress, userEmail) {
  try {
    const formData = new FormData();
    formData.append("file", file);
    selectedStems.forEach(stem => formData.append("stems", stem));

    const res = await axios.post(`${API_BASE}/upload`, formData, {
      headers: {
        "Content-Type": "multipart/form-data",
        ...(userEmail && { "X-User-Email": userEmail })
      },
      onUploadProgress: (e) => {
        if (onProgress && e.total) {
          const percent = Math.round((e.loaded * 100) / e.total);
          onProgress(percent);
        }
      }
    });

    return res.data;
  } catch (err) {
    throw new Error(err.response?.data?.error || "Upload failed.");
  }
}


export function streamJob(jobId, onMessage) {
  const evtSource = new EventSource(`${API_BASE}/stream/${jobId}`);
  evtSource.onmessage = (e) => {
    onMessage(e.data);
    if (e.data.includes("Job complete") || e.data.includes("Error")) {
      evtSource.close(); // ✅ Stop the stream after finishing
    }
  };
  return evtSource;
}

export async function downloadZip(jobId, onError) {
  try {
    const res = await axios.get(`${API_BASE}/download/${jobId}`, {
      responseType: "blob"
    });

    let filename = "download.zip";
    const header = res.headers["content-disposition"];
    if (header && header.includes("filename=")) {
      filename = header.split("filename=")[1].replace(/"/g, "");
    }

    const url = window.URL.createObjectURL(new Blob([res.data]));
    const link = document.createElement("a");
    link.href = url;
    link.setAttribute("download", filename);
    document.body.appendChild(link);
    link.click();
    link.remove();
    window.URL.revokeObjectURL(url);

  } catch (err) {
    if (onError) onError("Download failed. Please try again.");
  }
}

import React, { useState, useEffect, useRef } from "react";
import WaveSurfer from "wavesurfer.js";
import { uploadFile, streamJob, downloadZip } from "../api";
import axios from "axios";
import { useContext } from "react";
import { AuthContext } from "../context/AuthContext.jsx";

export default function UploadForm() {
  const [file, setFile] = useState(null);
  const [stems, setStems] = useState(["vocals", "other"]);
  const [status, setStatus] = useState("");
  const [jobId, setJobId] = useState(null);
  const [done, setDone] = useState(false);
  const [showPreview, setShowPreview] = useState(true);
  const [uploading, setUploading] = useState(false);
  const [evtSource, setEvtSource] = useState(null);
  const [error, setError] = useState("");
  const waveRefs = useRef({});
  const [playingStates, setPlayingStates] = useState({});
  const [volumes, setVolumes] = useState({});
  const { user } = useContext(AuthContext);

  // ✅ Usage state
  const [usedMB, setUsedMB] = useState(0);
  const [limitMB, setLimitMB] = useState(150);

  const fetchUsage = async () => {
    try {
      const res = await axios.get("http://127.0.0.1:5000/usage", {
        headers: {
          ...(user?.email && { "X-User-Email": user.email })
        }
      });
      setUsedMB(res.data.used_mb);
      setLimitMB(res.data.limit_mb);
    } catch (err) {
      console.error("Failed to fetch usage", err);
    }
  };

  useEffect(() => {
    fetchUsage();
  }, []);

  const toggleStem = (stem) => {
    setStems((prev) =>
      prev.includes(stem) ? prev.filter((s) => s !== stem) : [...prev, stem]
    );
  };

  const startUpload = async (selectedFile) => {
    if (!selectedFile) return;
    setFile(selectedFile);
    setUploading(true);
    setStatus("Uploading...");
    setDone(false);
    setShowPreview(true);
    setError("");

    console.log("Logged in as:", user?.email); 

    let res;
    try {
      res = await uploadFile(selectedFile, stems, null, user?.email);
      if (!res?.job_id) throw new Error("No job ID received.");
    } catch (err) {
      console.error(err);
      setError(err.message || "Upload failed. Please try again.");
      setUploading(false);
      return;
    }

    setJobId(res.job_id);
    setStatus("Processing...");

    const es = streamJob(res.job_id, (msg) => {
      setStatus(msg);
      if (msg.includes("Job complete")) {
        setTimeout(() => {
          setDone(true);
          setUploading(false);
          fetchUsage(); // ✅ update usage after job finishes
        }, 500);
      }
    });

    setEvtSource(es);
  };

  const handleDownload = () => {
    if (jobId) {
      downloadZip(jobId, (errMsg) => setError(errMsg));
      setFile(null);
      setJobId(null);
      setDone(false);
      setStatus("");
      setShowPreview(false);
      setStems(["vocals", "other"]);
      setUploading(false);
      setPlayingStates({});
      setVolumes({});
    }
  };

  const handleFileSelect = (e) => {
    console.log("FILE SELECTED");
    const selectedFile = e.target.files[0];
    if (selectedFile) startUpload(selectedFile);
  };

  const handleDrop = (e) => {
    e.preventDefault();
    const droppedFile = e.dataTransfer.files[0];
    if (droppedFile) startUpload(droppedFile);
  };

  const handleDragOver = (e) => {
    e.preventDefault();
  };

  useEffect(() => {
    return () => {
      if (evtSource) evtSource.close();
      Object.values(waveRefs.current).forEach((ws) => ws.destroy());
    };
  }, [evtSource]);

  const createWaveform = (url, stem) => {
    const container = document.getElementById(`waveform-${stem}`);
    if (!container) return;

    if (waveRefs.current[stem]) {
      waveRefs.current[stem].destroy();
    }

    waveRefs.current[stem] = WaveSurfer.create({
      container: container,
      waveColor: "#4F46E5",
      progressColor: "#22C55E",
      cursorColor: "#FFFFFF",
      height: 80,
      barWidth: 2,
      responsive: true,
      volume: volumes[stem] ?? 1,
    });

    waveRefs.current[stem].load(url);
  };

  useEffect(() => {
    if (done && showPreview && jobId) {
      stems.forEach((stem) => {
        const url = `http://127.0.0.1:5000/preview/${jobId}/${stem}`;
        createWaveform(url, stem);
      });
    }
  }, [done, showPreview, jobId]);

  const toggleStemPlay = (stem) => {
    const ws = waveRefs.current[stem];
    if (!ws) return;
    if (playingStates[stem]) {
      ws.pause();
      setPlayingStates((prev) => ({ ...prev, [stem]: false }));
    } else {
      ws.play();
      setPlayingStates((prev) => ({ ...prev, [stem]: true }));
    }
  };

  const handleVolumeChange = (stem, value) => {
    const vol = Math.max(0, Math.min(1, value));
    setVolumes((prev) => ({ ...prev, [stem]: vol }));
    if (waveRefs.current[stem]) {
      waveRefs.current[stem].setVolume(vol);
    }
  };

  const startDrag = (stem, e) => {
    const bar = e.currentTarget;

    const updateVolume = (event) => {
      const rect = bar.getBoundingClientRect();
      const y = event.clientY - rect.top;
      const height = rect.height;
      const newVol = 1 - y / height;
      handleVolumeChange(stem, newVol);
    };

    updateVolume(e);

    const moveHandler = (event) => updateVolume(event);
    const upHandler = () => {
      window.removeEventListener("mousemove", moveHandler);
      window.removeEventListener("mouseup", upHandler);
    };

    window.addEventListener("mousemove", moveHandler);
    window.addEventListener("mouseup", upHandler);
  };

  const usagePercent = Math.min(100, Math.round((usedMB / limitMB) * 100));

  return (
    <div
      className="bg-gray-800 bg-opacity-50 p-6 rounded-xl shadow-lg"
      onDrop={handleDrop}
      onDragOver={handleDragOver}
    >
      {/* ✅ USAGE BAR */}
      <div className="mb-4">
        <p className="text-sm text-gray-200 mb-1">
          📊 Usage: {usedMB}MB / {limitMB}MB ({usagePercent}%)
        </p>
        <div className="w-full h-3 bg-gray-600 rounded">
          <div
            className={`h-3 rounded transition-all duration-300 ${
              usagePercent < 60
                ? "bg-green-500"
                : usagePercent < 90
                ? "bg-yellow-400"
                : "bg-red-500"
            }`}
            style={{ width: `${usagePercent}%` }}
          ></div>
        </div>
      </div>

      {/* ERROR */}
      {error && (
        <div className="mb-4 p-3 bg-red-600 bg-opacity-80 rounded text-white text-sm">
          {error}
        </div>
      )}

      {/* UPLOAD + STEMS */}
      {!done && !uploading && (
        <>
          <label className="flex flex-col items-center justify-center w-full h-32 border-2 border-dashed border-gray-500 rounded-lg cursor-pointer hover:border-indigo-400 transition">
            <p className="text-gray-300 text-sm mb-2">
              Drag & Drop or Click to Select a File
            </p>
            <input
              type="file"
              onChange={handleFileSelect}
              className="hidden"
            />
          </label>

          <div className="flex flex-wrap gap-4 justify-center mt-4">
            {["vocals", "drums", "bass", "other"].map((stem) => (
              <label key={stem} className="text-sm">
                <input
                  type="checkbox"
                  className="mr-2"
                  checked={stems.includes(stem)}
                  onChange={() => toggleStem(stem)}
                  disabled={uploading || jobId !== null}
                />
                {stem.charAt(0).toUpperCase() + stem.slice(1)}
              </label>
            ))}
          </div>
        </>
      )}

      {/* LOADING */}
      {uploading && !done && (
        <div className="flex flex-col items-center justify-center mt-8">
          <div className="flex space-x-1 h-8">
            {[...Array(5)].map((_, i) => (
              <div
                key={i}
                className="w-2 bg-indigo-500 rounded animate-pulse"
                style={{
                  animationDelay: `${i * 0.15}s`,
                  animationDuration: "1s",
                  height: "100%",
                  transformOrigin: "center bottom",
                  animationName: "wavePulse",
                }}
              />
            ))}
          </div>
          <p className="mt-4 text-gray-300 text-sm">{status}</p>

          <style>
            {`
              @keyframes wavePulse {
                0%, 100% { transform: scaleY(0.3); opacity: 0.5; }
                50% { transform: scaleY(1); opacity: 1; }
              }
              .animate-pulse {
                animation: wavePulse 1s ease-in-out infinite;
              }
            `}
          </style>
        </div>
      )}

      {/* DOWNLOAD + PREVIEW */}
      {done && (
        <div className="mt-6">
          <button
            onClick={handleDownload}
            className="bg-green-600 hover:bg-green-700 px-6 py-2 rounded-lg font-medium"
          >
            Download Stems
          </button>

          {showPreview && (
            <div className="mt-6 bg-gray-900 p-4 rounded-lg">
              <h3 className="text-white mb-2">Preview Stems</h3>
              {stems.map((stem) => (
                <div key={stem} className="mb-6 flex items-center space-x-3">
                  <div className="min-w-[60px] text-sm text-white font-medium capitalize">
                    {stem}
                  </div>

                  <button
                    onClick={() => toggleStemPlay(stem)}
                    className="bg-gray-700 hover:bg-gray-600 p-2 rounded-full flex items-center justify-center"
                  >
                    <svg
                      xmlns="http://www.w3.org/2000/svg"
                      className="w-6 h-6 text-white"
                      fill="currentColor"
                      viewBox="0 0 24 24"
                    >
                      {playingStates[stem] ? (
                        <path d="M6 4h4v16H6zm8 0h4v16h-4z" />
                      ) : (
                        <path d="M8 5v14l11-7z" />
                      )}
                    </svg>
                  </button>

                  <div
                    id={`waveform-${stem}`}
                    className="flex-1 bg-black rounded"
                  ></div>

                  <div
                    className="relative w-8 h-32 bg-gray-700 rounded-lg cursor-pointer flex flex-col justify-end items-center overflow-hidden"
                    onMouseDown={(e) => startDrag(stem, e)}
                  >
                    <div
                      className="absolute bottom-0 left-0 right-0 rounded-lg transition-all duration-200"
                      style={{
                        height: `${(volumes[stem] ?? 1) * 100}%`,
                        background: "linear-gradient(to top, #22C55E, #4F46E5)",
                      }}
                    ></div>
                    <span className="relative text-xs text-white font-bold mb-1 z-10">
                      {Math.round((volumes[stem] ?? 1) * 100)}
                    </span>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      )}
    </div>
  );
}

## 📄 frontend/src/context/AuthContext.jsx
import React from "react";
import { createContext, useState, useEffect } from "react";

export const AuthContext = createContext();

export function AuthProvider({ children }) {
    const [user, setUser] = useState(() => {
        try {
            const stored = localStorage.getItem("user");
            return stored ? JSON.parse(stored) : null;
        } catch {
            return null;
        }
    });

    const login = (email) => {
        const newUser = { email };
        setUser(newUser);
        localStorage.setItem("user", JSON.stringify(newUser));
    };

    const logout = () => {
        setUser(null);
        localStorage.removeItem("user");
    };

    return (
        <AuthContext.Provider value={{ user, login, logout }}>
            {children}
        </AuthContext.Provider>
    );
}
